' One click. Every report. Zero copy-paste.
Sub BuildMonthlyReport()
    Dim ws As Worksheet, rows As Long
    Application.ScreenUpdating = False

    Set ws = Refresh("RawData")
    rows = CleanAndValidate(ws)
    BuildPivots ws
    ExportPdf "Report_" & Format(Date, "yyyy-mm")

    MsgBox rows & " rows processed in 4s", vbInformation
End Sub
